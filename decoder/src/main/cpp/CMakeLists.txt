cmake_minimum_required(VERSION 3.22.1)
project(rezka_mpeg4_decoder LANGUAGES C CXX)

set(FFMPEG_LIBS "${CMAKE_CURRENT_SOURCE_DIR}/../jniLibs/${ANDROID_ABI}")
foreach(lib avutil avcodec swscale)
    file(GLOB candidates "${FFMPEG_LIBS}/lib${lib}.so*")
    if(NOT candidates)
        message(FATAL_ERROR "Missing FFmpeg ${lib} for ${ANDROID_ABI}; run build_ffmpeg_android.sh first")
    endif()
    list(GET candidates 0 candidate)
    add_library(${lib} SHARED IMPORTED)
    set_target_properties(${lib} PROPERTIES IMPORTED_LOCATION "${candidate}")
endforeach()

add_library(rezka_mp4v SHARED mp4v_jni.cpp)
target_include_directories(rezka_mp4v PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../ffmpeg-headers")
target_link_options(rezka_mp4v PRIVATE "-Wl,-z,max-page-size=16384")
find_library(android-lib android)
find_library(log-lib log)
target_link_libraries(rezka_mp4v avcodec swscale avutil ${android-lib} ${log-lib})
